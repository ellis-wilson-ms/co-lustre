--- Description ---
This issue can be reproduced by the following steps:

lctl set_param mdd.*.trash_can_enable=1
lctl set_param mdd.*.trash_can_type=uid
mkdir -p /mnt/lustre/d
touch /mnt/lustre/d/f
rm /mnt/lustre/d/f
cd /mnt/lustre/.ltrash/MDT0000/0
rm $pfid/f
rm -r $pfid
