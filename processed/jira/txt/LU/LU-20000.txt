--- Description ---
CPTs can be set for each NI, but because of a bug, all NIs in a net will have the same CPTs.

Also, tunables parsing is wrong. But for this issue, there are other problems...

In handle_yaml_config_ni header comment, we have:

net:
   - net type: []
     local NI(s):
       - nid: @[]
         status: up
         interfaces:
             0: ['['']']
             1: ['['']']
      tunables:
             peer_timeout: 
             peer_credits: 
             peer_buffer_credits: 
             credits: 
      lnd tunables:
             peercredits_hiw: 
             map_on_demand: 
             concurrent_sends: 
             fmr_pool_size: 
             fmr_flush_trigger: 
             fmr_cache: 

However, with export and net show, it is this format:

net:
   - net type: []
     local NI(s):
       - nid: @[]
         status: up
         interfaces:
             0: ['['']']
             1: ['['']']
         tunables:
                peer_timeout: 
                peer_credits: 
                peer_buffer_credits: 
                credits: 
         lnd tunables:
                peercredits_hiw: 
                map_on_demand: 
                concurrent_sends: 
                fmr_pool_size: 
                fmr_flush_trigger: 
                fmr_cache:

net:
