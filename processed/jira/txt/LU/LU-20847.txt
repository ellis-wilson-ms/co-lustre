test_630() { 
 reinit_dlc || return $? 
 local output 
 local rc 
 output=$($LNETCTL peer show --nid 1.1.1.1@tcp 2>&1)
 rc=$? 
 echo "rc=$rc output='$output'" 
 ((rc != 0)) || error "peer show of an unknown nid should fail" 
 echo "$output" | grep -q "cannot find peer" || 
         error "Bad or missing error message, got: '$output'" 
} 
run_test 630 "peer show: unknown peer should be reported"
