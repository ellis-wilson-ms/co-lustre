--- Description ---
In lustre/ptlrpc/niobuf.c, kick_cpu_latency() takes latency_qos->lock before allocating a struct dev_pm_qos_request.  When OBD_ALLOC_PTR() fails, the function warns and returns immediately without dropping the mutex:

                mutex_lock(&latency_qos->lock);
                if (latency_qos->pm_qos_req == NULL) {
                        OBD_ALLOC_PTR(latency_qos->pm_qos_req);
                        if (latency_qos->pm_qos_req == NULL) {
                                CWARN("%s: Failed to allocate a PM-QoS request for cpu %d\n",
                                      obd->obd_name, cpu);
                                return;
                        }
