--- Description ---
tunefs.lustre --quota enables quota accounting on the backing filesystem and then returns immediately, so osd_write_ldd() is never reached:

        /* Enable quota accounting */
        if (mop.mo_flags & MO_QUOTA) {
                ret = osd_enable_quota(&mop);
                goto out;
        }write:#endif /* !TUNEFS */        /* Write our config files */
        ret = osd_write_ldd(&mop);
        if (ret != 0) {
                fatal();
                fprintf(stderr, "failed to write local files\n");
                goto out;
        }
out:
        osd_fini();
        ret2 = loop_cleanup(&mop);
        if (ret == 0)
                ret = ret2;
