as non-root user test1

[test1@trevis-90vm1 test1]$ lctl get_param version
version=2.16.0_ddn44_15_g431bf69

[test1@trevis-90vm1 /]$ cd /mnt/lustre

[test1@trevis-90vm1 lustre]$ cd test1/
[test1@trevis-90vm1 test1]$ ls -al

[test1@trevis-90vm1 test1]$ rm foo 

[test1@trevis-90vm1 test1]$ ls .Trash
