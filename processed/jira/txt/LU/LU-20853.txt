--- Description ---
systemd.pc is shipped in systemd-dev on Ubuntu 26.04. Without it, configure silently disables systemd support, and packages built by make debs/make dkms-debs do not include lnet.service or the other unit files.

root@ubuntu2604:~/fs-lustre-release# dpkg -l | grep systemd-dev

root@ubuntu2604:~/fs-lustre-release# pkg-config --variable=systemdsystemunitdir systemd

root@ubuntu2604:~/fs-lustre-release# ./autogen.sh && ./configure --disable-server --with-o2ib=/usr/src/ofa_kernel/x86_64/7.0.0-31-generic && make && make dkms-debs
...

root@ubuntu2604:~/fs-lustre-release# cd debs/
root@ubuntu2604:~/fs-lustre-release/debs# dpkg -c lustre-client-utils_2.17.58-39-g094031c-1_amd64.deb | grep system
