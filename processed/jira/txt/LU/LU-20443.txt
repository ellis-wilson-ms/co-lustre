void ll_deauthorize_statahead(struct inode *dir, void *key)                   
{                                                                             
        struct ll_inode_info *lli = ll_i2info(dir);                           
        struct ll_statahead_info *sai;                                        
                                                                              
        LASSERT(lli->lli_opendir_key == key);                                 
        LASSERT(lli->lli_opendir_pid != 0);                                   
                                                                              
        printk("ll_deauthorize_statahead: %p\n", dir);                        
        CDEBUG(D_READA, "deauthorize statahead for "DFID"\n",                 
               PFID(&lli->lli_fid));                                          
                                                                              
        spin_lock(&lli->lli_sa_lock);                                         
        lli->lli_opendir_key = NULL;                                          
        lli->lli_opendir_pid = 0;                                             
        lli->lli_sa_enabled = 0;                                              
        sai = lli->lli_sai;                                                   
        if (sai && sai->sai_task) {                                           
                /*                                                            
                 * statahead thread may not have quit yet because it needs to 
                 * cache entries, now it's time to tell it to quit.           
                 *                                                            
                 * wake_up_process() provides the necessary barriers          
                 * to pair with set_current_state().                          
                 */                                                           
                struct task_struct *task = sai->sai_task;                     
                                                                              
                /* matches smp_load_acquire() in ll_statahead_thread() */     
                smp_store_release(&sai->sai_task, NULL);                      
                wake_up_process(task);                                        
        }                                                                     
        spin_unlock(&lli->lli_sa_lock);                                       
        CFS_RACE(OBD_FAIL_LLITE_STATAHEAD_RACE);                              
}                                                                             

This seems to be a race between start_statahead_thread() and ll_deauthorize_statahead().

	One thread try to start statahead thread: sai_task is not yet set.

	The other close the FD: ll_deauthorize_statahead(). It resets lli_opendir_key/lli_opendir_pid but could not wake_up the statahead thread (not yet started).

	Once statahead starts and sai_task is set, the thread remains blocked and never wakes up.
